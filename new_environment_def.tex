%--------------ordinary font size--------------
\newcommand{\ordinaryfontsize}{\fontsize{12pt}{24pt}\selectfont} % set the ordinary font size

%--------------section--------------
\newcommand{\newsection}[1]{
\fontsize{16pt}{24pt}\selectfont
\noindent{\color{gray}\rule{\linewidth}{0.4pt}}

    \vspace{-0.2cm}

\noindent\textsc{\textbf{#1}}
\ordinaryfontsize % back to the ordinary font size
\vspace{-0.5cm}
}

%--------------major item (with time)--------------
% \resemeitem{<experience>}{<date>}
\newcommand{\resemeitem}[2]{
\vspace{0.5mm}\item

    {\color{violet}\textbf{#1}} \hfill {\fontsize{9pt}{10pt}\selectfont
        \color{gray}\textbf{\textit{#2}}}
    
    \vspace{-5mm}
}

\newenvironment{resemelist}
{
  \begin{list}{}{ % label
    \setlength{\leftmargin}{1mm}
    \setlength{\itemsep}{1mm}
  }
}
{
  \end{list}
  
  %\vspace{-5mm}
}
